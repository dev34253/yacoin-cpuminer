function(yac_add_test name)
  add_executable(${name} ${name}.cpp)
  target_link_libraries(${name} PRIVATE yac_core)
  target_include_directories(${name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_compile_definitions(${name} PRIVATE YAC_TEST_DATA_DIR="${CMAKE_CURRENT_SOURCE_DIR}/data")
  add_test(NAME ${name} COMMAND ${name})
endfunction()

yac_add_test(test_util)
yac_add_test(test_target)
yac_add_test(test_getwork)
yac_add_test(test_rpc)
yac_add_test(test_miner)
yac_add_test(test_hash)
# test_hash runs real N-factor-21 hashes (512 MiB, a few seconds)
set_tests_properties(test_hash PROPERTIES TIMEOUT 600)
